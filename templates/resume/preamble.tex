%----------CUSTOM COMMANDS----------
% Main resume item
\newcommand{\resumeItem}[1]{
  \item{\small #1\par}\vspace{\globalItemSep}
}

% Subheading with roles and dates
\newcommand{\resumeSubheading}[4]{
  \vspace{\globalSubsectionSep}\item
    \begin{tabular*}{\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} $|$ \textbf{#3} & \small#2 $|$ \textit{\small #4} \\
    \end{tabular*}\vspace{-15pt}
}

% Subheading within sections
\newcommand{\resumeSubSubheading}[2]{
    \item
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \textit{\small#1} & \textit{\small #2} \\
    \end{tabular*}\vspace{-7pt}
}

% Project Heading
\newcommand{\resumeProjectHeading}[2]{
    \vspace{\globalSubsectionSep}\item
    \begin{tabular*}{\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \small#1 & #2 \\
    \end{tabular*}\vspace{-15pt}
}

% Subitem (secondary bullet point)
\newcommand{\resumeSubItem}[1]{\resumeItem{#1}\vspace{\globalSubItemSep}}

% List environments using global parameters
\newcommand{\resumeSubHeadingListStart}{
  \begin{itemize}[
    leftmargin=0in,
    topsep=0pt,
    label={}
  ]
}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}

\newcommand{\resumeItemListStart}{
  \begin{itemize}[
    align=left,
    leftmargin=\globalLeftMargin,
    labelsep=\globalLabelSep
  ]
}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace{-7pt}}
